(after! org-roam
  ;; md-roam
  (require 'md-roam)
  (setq md-roam-file-extension "md")
  (md-roam-mode 1)
  (setq org-roam-file-extensions '("org" "md"))

  (defun tsid (&optional time)
    "Return timestamp-id."
    (format-time-string "%Y%m%d%H%M%S" time "UTC"))

  ;; org-roam-capture--setup-target-location зовёт `org-entry-get p "ID"`
  ;; на свежесозданном буфере. В markdown это уходит в org-element parser,
  ;; который падает на rx (** 1 0 "*"). md-roam держит ID во frontmatter
  ;; и сам отвечает за org-id-get; здесь просто говорим «нет org-property».
  (defadvice! my/org-entry-get-skip-md-a (fn &rest args)
    :around #'org-entry-get
    (unless (eq major-mode 'markdown-mode)
      (apply fn args)))

  ;; Doom :editor file-templates вставляет дефолтный snippet «# Document Title\n»
  ;; в любой пустой .md и активирует yas-snippet. org-roam-capture сразу пишет
  ;; свой frontmatter поверх, snippet остаётся с висящими маркерами и yas
  ;; падает в post-command-hook. Отключаем file-templates для roam-каталога.
  (when (fboundp 'set-file-template!)
    (set-file-template! (concat "\\`" (regexp-quote (file-truename org-roam-directory)))
      :ignore t))

  ;; Шаблоны
  (defconst slip-boxes
    '(("d" "default"      ""            "${capture-tsid}" "org" "#+filetags: review")
      ("m" "markdown"     ""            "${capture-tsid}" "md"  nil)
      ("a" "areas"        "areas/"      "${citekey}"      "org" nil)
      ("l" "literature"   "literature/" "${capture-tsid}" "org"
       "#+filetags: review\n#+LAST_MODIFIED: \n#+DATE: %<%Y-%m-%d>\n"))
    "My slip boxes.")

      (setq org-roam-capture-templates
            (mapcar (lambda (x)
                      (let* ((key            (nth 0 x))
                            (name           (nth 1 x))
                            (dir            (nth 2 x))
                            (filename       (nth 3 x))
                            (ext            (nth 4 x))
                            (extra-template (nth 5 x))
                            (is-md          (string= ext "md"))
                            (header (if is-md
                                        (concat "---\n"
                                                "title: ${title}\n"
                                                "id: %<%Y%m%d%H%M%S>\n"
                                                "---\n\n"
                                                "# ${title}\n\n")
                                      (concat "#+TITLE: ${title}\n"
                                              (or extra-template "")))))
                        `(,key ,name plain "%?"
                          :if-new (file+head
                                  ,(concat dir filename "." ext)
                                  ,header)
                          :immediate-finish t
                          :unnarrowed t)))
                    slip-boxes))

  (defun capture-tsid (node)
    "A hack definition to workaround that org-roam passes a node argument."
    (tsid))

  (defun org-roam-new-node (&optional keys)
    (interactive)
    (org-roam-capture-
     :keys keys
     :node (org-roam-node-create :title "")
     :props '(:finalize find-file)))

  (defun org-roam-new-node-default ()
    "Create a new default org-mode roam note."
    (interactive)
    (org-roam-new-node "d"))

  (defun org-roam-new-node-areas ()
    "Create a new note in areas/ slip-box."
    (interactive)
    (org-roam-new-node "a"))

  (defun org-roam-new-node-literature ()
    "Create a new literature note in literature/ slip-box."
    (interactive)
    (org-roam-new-node "l"))

  (defun org-roam-new-node-markdown ()
    "Create a new markdown roam note."
    (interactive)
    (org-roam-new-node "m"))

  (map! :leader
        (:prefix ("n r" . "roam")
         (:prefix ("N" . "new note")
          :desc "Default (org)"  "d" #'org-roam-new-node-default
          :desc "Literature"     "l" #'org-roam-new-node-literature
          :desc "Markdown"       "m" #'org-roam-new-node-markdown
          :desc "Areas"          "a" #'org-roam-new-node-areas)))

  (defun move-to-slip-box (slip-box)
    "Move file to specified SLIP-BOX."
    (interactive (list (completing-read "Move to slip-box: "
                                        (mapcar (lambda (x) (nth 2 x)) slip-boxes))))
    (let* ((filename (buffer-file-name))
           (directory (file-name-directory filename))
           (name (file-name-nondirectory filename))
           (new-name (f-join org-roam-directory slip-box name)))
      (roam-rename new-name)))

  (defun roam-rename (new-name)
    "Move file to NEW-NAME."
    (let ((filename (buffer-file-name)))
      (unless filename
        (error "Buffer '%s' is not visiting file!" (buffer-name)))
      (rename-file filename new-name)
      (set-visited-file-name new-name t)
      (revert-buffer t t t)
      (set-buffer-modified-p t)
      (save-buffer)))

  (setq org-roam-node-display-template "${hierarchy:*} ${tags:10}")

  ;; ВОТ КЛЮЧЕВОЕ ИСПРАВЛЕНИЕ — eval вместо прямого cl-defmethod
  (eval
   '(progn
      (cl-defmethod org-roam-node-filetitle ((node org-roam-node))
        "Return the file TITLE for the node."
        (org-roam-get-keyword "TITLE" (org-roam-node-file node)))

      (cl-defmethod org-roam-node-hierarchy ((node org-roam-node))
        "Return the hierarchy for the node."
        (let ((title (org-roam-node-title node))
              (olp (org-roam-node-olp node))
              (level (org-roam-node-level node))
              (directories (org-roam-node-directories node))
              (filetitle (org-roam-node-filetitle node)))
          (concat
           (if directories (format "(%s) " directories))
           (if (> level 0) (concat filetitle " > "))
           (if (> level 1) (concat (string-join olp " > ") " > "))
           title)))

      (cl-defmethod org-roam-node-directories ((node org-roam-node))
        (if-let ((dirs (file-name-directory
                        (file-relative-name
                         (org-roam-node-file node) org-roam-directory))))
            (string-join (f-split dirs) "/")
          nil)))))
