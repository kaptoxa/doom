(defun my/org-roam-moc-stats ()
  "Показать MOC-заметки и число исходящих ссылок."
  (interactive)
  (let* ((moc-nodes (seq-filter
                     (lambda (node)
                       (member "moc" (org-roam-node-tags node)))
                     (org-roam-node-list)))
         (stats (mapcar
                 (lambda (node)
                   (let* ((id (org-roam-node-id node))
                          (links (org-roam-db-query
                                  [:select (funcall count dest)
                                   :from links
                                   :where (= source $s1)
                                   :and (= type "id")]
                                  id)))
                     (cons (org-roam-node-title node)
                           (caar links))))
                 moc-nodes))
         (sorted (seq-sort-by #'cdr #'> stats)))
    (with-current-buffer (get-buffer-create "*MOC Stats*")
      (erase-buffer)
      (insert "MOC заметка | Исходящих ссылок\n")
      (insert (make-string 50 ?-) "\n")
      (dolist (s sorted)
        (insert (format "%-40s %d\n" (car s) (cdr s))))
      (switch-to-buffer (current-buffer)))))

    (let* ((entry (car (auth-source-search
                        :host connection
                        :max 1
                        :require '(:secret))))

** LSP
#+begin_src emacs-lisp
(setq read-process-output-max (* 1024 1024)) ; 1mb — меньше лагов при большом выводе lsp
(after! lsp-mode
  (setq lsp-enable-file-watchers nil    ; file-watchers через tramp не работают/дороги
        lsp-log-io nil))

;; На VM стоит только pylsp (см. codenv/setup.d/40-install-lsp-server), а pyright
;; там нет. lsp-mode клонирует каждый клиент в "-tramp"-версию с тем же приоритетом,
;; поэтому remote-клиент pyright-tramp (priority 2) перебивает pylsp-tramp (priority -1)
;; и лезет за несуществующим на VM pyright-langserver. Отключаем pyright на remote.
(add-hook! '(python-mode-local-vars-hook python-ts-mode-local-vars-hook)
  (defun +lsp-prefer-pylsp-on-remote-h ()
    (when (file-remote-p default-directory)
      (setq-local lsp-disabled-clients '(pyright pyright-remote pyright-tramp)))))
#+end_src
